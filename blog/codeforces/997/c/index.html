代码

#include <iostream>
#include <algorithm>
#include <cstdio>
#include <cstring>
#include <cmath>

using namespace std;

typedef long long ll;

const int N = 1e6+10;
const ll MOD = 998244353;

ll n;
ll pow3[N] , fac[N] , invfac[N];

ll qmi(ll a , ll b) {
	ll res = 1;
	while(b) {
		if(b & 1) res = res * a % MOD;
		a = a * a % MOD; b >>= 1;
	}
	return res;
}

void add(ll &a , ll b) {
	a += b;
	if(a >= MOD) a -= MOD;
	else if(a < 0) a += MOD;
}

ll C(ll x , ll y) {
	return fac[x] * invfac[y] % MOD * invfac[x-y] % MOD;
}

void init() {
	pow3[0] = 1; fac[0] = 1;
	for(int i = 1 ; i <= n ; i ++) {
		pow3[i] = pow3[i-1] * 3ll % MOD;
		fac[i] = fac[i-1] * i % MOD;
	}
	invfac[n] = qmi(fac[n] , MOD-2);
	for(int i = n ; i >= 1 ; i --) {
		invfac[i-1] = invfac[i] * i % MOD;
	}
}

int main() {
	scanf("%lld" , &n);
	init();
	
	ll ans = qmi(3 , n*n+1) , res = 0;
	for(int i = 1 ; i <= n ; i ++) {
		add(res , C(n,i) * (i&1 ? -1 : 1) * ((qmi(1-pow3[i] * qmi(pow3[n], MOD-2) % MOD , n)-1) % MOD + MOD) % MOD
			* qmi(qmi(3,n*i) , MOD-2) % MOD);
	}
	ans = ans * res % MOD;
	add(ans , 2ll * qmi(3,n*n) % MOD * ((qmi(1-qmi(pow3[n-1],MOD-2) , n)-1)%MOD + MOD) % MOD);
	ans = (MOD-ans)%MOD;
	printf("%lld\n" , ans);
	return 0;
}
